class Dirs:
	"""
	Класс-обёртка для колонок таблицы `dirs` в базе данных.
	Предоставляет удобный доступ к колонкам через атрибуты класса.
	"""
	table = _table_dirs
	id = _table_dirs.c.id
	rel_dir_path = _table_dirs.c.short_src
	mod = _table_dirs.c.mod
	mf_alias = _table_dirs.c.brand

	@classmethod
	def reset_mod(cls, mf_alias: str):
		"""
		Сбрасывает mod всех папок Mf в 0, чтобы сканер при следующем запуске
		сравнил содержимое каждой папки, а не пропустил её по совпадению mtime.
		"""
		with Dbase.main_engine.begin() as conn:
			stmt = (
				sqlalchemy.update(cls.table)
				.where(cls.mf_alias == mf_alias)
				.values({cls.mod.name: 0})
			)
			conn.execute(stmt)

Как должен работать сканер

берется mf_alias и mf_path
и базы данных из DIRS берутся ВСЕ записи соответствующие mf_alias
у нас получится список оносительных путей + mod + mf_alias

из Finder scandir берутся все директории по mf_path
затем это превращается в список относительных путей
превращение = убираем из абсолютного пути mf_path

Получается мы будем сравнивать списки относительных путей и так мы узнаем:
какие директории изменились удалились добавились и уже оттталикваясь от этого далее работаем
